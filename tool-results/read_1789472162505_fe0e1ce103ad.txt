     4→import { toast } from 'sonner'
     5→import {

     9→  ChevronRight,

    23→import {

    27→  DialogTitle,
    28→} from '@/components/ui/dialog'
    29→import { Input } from '@/components/ui/input'

    34→  formatKm,

    36→  isTaxAlive,
    37→  waLink,
    38→} from '@/lib/format'
    39→import {

    41→  loadMarketingSession,
    42→  saveMarketingSession,
    43→} from '@/lib/marketing-session'
    44→import type {
    45→  MarketingSession,
    46→  PublicCatalogResponse,
    47→  PublicVehicle,
    48→  TaxonomyResponse,
    49→} from '@/lib/types'
    50→
    51→export type StatusFilter = 'all' | 'available' | 'hold'
    52→
    53→export function CatalogClient({ slug }: { slug: string }) {
    54→  const [data, setData] = useState<PublicCatalogResponse | null>(null)
    55→  const [loading, setLoading] = useState(true)
    56→  const [error, setError] = useState<string | null>(null)
    57→  const [notFound, setNotFound] = useState(false)
    58→
    59→  // Sistem Rekanan Terdaftar: sesi verifikasi WhatsApp marketing
    60→  const [marketing, setMarketing] = useState<MarketingSession | null>(null)
    61→  const [phase, setPhase] = useState<'checking' | 'gate' | 'open'>('checking')
    62→  const marketingRef = useRef<MarketingSession | null>(null)
    63→
    64→  const [statusFilter, setStatusFilter] = useState<StatusFilter>('all')
    65→  const [brandFilter, setBrandFilter] = useState<string>('all')
    66→  const [categoryFilter, setCategoryFilter] = useState<string>('all')
    67→  const [branchFilter, setBranchFilter] = useState<string>('all') // all | main | branchId
    68→  const [q, setQ] = useState('')
    69→  const [holdTarget, setHoldTarget] = useState<PublicVehicle | null>(null)
    70→  const [gallery, setGallery] = useState<{ vehicle: PublicVehicle; index: number } | null>(null)
    71→  const [taxonomy, setTaxonomy] = useState<TaxonomyResponse | null>(null)
    72→
    73→  const refetch = useCallback(
    74→    async (silent = false, session?: MarketingSession | null) => {
    75→      if (!silent) setLoading(true)
    76→      try {
    77→        const mkt = session ?? marketingRef.current
    78→        const res = await fetch(`/api/showrooms/${slug}/vehicles`, {
    79→          cache: 'no-store',
    80→          ...(mkt ? { headers: { 'X-Mkt-Phone': mkt.phone } } : {}),
    81→        })
    82→        if (res.status === 404) {
    83→          setNotFound(true)
    84→          return
    85→        }
    86→        if (res.status === 403) {
    87→          const j = await res.json().catch(() => ({}))
    88→          if (j.code === 'WHITELIST_REQUIRED') {
    89→            // Nomor sesi tidak lagi terdaftar/aktif — paksa verifikasi ulang
    90→            clearMarketingSession(slug)
    91→            marketingRef.current = null
    92→            setMarketing(null)
    93→            setPhase('gate')
    94→            return
    95→          }
    96→          throw new Error(j.error || 'Akses katalog ditolak.')
    97→        }
    98→        if (!res.ok) {
    99→          const j = await res.json().catch(() => ({}))
   100→          throw new Error(j.error || 'Gagal memuat katalog.')
   101→        }
   102→        setData(await res.json())
   103→        setError(null)
   104→        setPhase('open')
   105→      } catch (e) {
   106→        if (!silent) setError(e instanceof Error ? e.message : 'Gagal memuat katalog.')
   107→      } finally {
   108→        if (!silent) setLoading(false)
   109→      }
   110→    },
   111→    [slug],
   112→  )
   113→
   114→  useEffect(() => {
   115→    // Baca sesi rekanan dari localStorage — marketing tidak perlu mengetik ulang nomor
   116→    const cached = loadMarketingSession(slug)
   117→    marketingRef.current = cached
   118→    setMarketing(cached)
   119→    refetch(false, cached)
   120→  }, [refetch, slug])
   121→
   122→  // Filter kategori & merek dari konfigurasi showroom (bisa ditambah owner)
   123→  useEffect(() => {
   124→    fetch(`/api/showrooms/${slug}/taxonomy`, { cache: 'no-store' })
   125→      .then((r) => (r.ok ? r.json() : null))
   126→      .then((j) => j && setTaxonomy(j))
   127→      .catch(() => {})
   128→  }, [slug])
   129→
   130→  // Auto-refresh: tahanan kedaluwarsa harus kembali Ready tanpa manual refresh
   131→  useEffect(() => {
   132→    const iv = setInterval(() => refetch(true), 30000)
   133→    const onVis = () => {
   134→      if (document.visibilityState === 'visible') refetch(true)
   135→    }
   136→    document.addEventListener('visibilitychange', onVis)
   137→    return () => {
   138→      clearInterval(iv)
   139→      document.removeEventListener('visibilitychange', onVis)
   140→    }
   141→  }, [refetch])
   142→
   143→  const brands = useMemo(() => {
   144→    // Gabung taxonomy (urutan owner) + nilai faktual di stok (antisipasi data lama)
   145→    const inStock = [...new Set((data?.vehicles ?? []).map((v) => v.brand))].sort()
   146→    const all = [...new Set([...(taxonomy?.brands ?? []), ...inStock])]
   147→    return all
   148→  }, [data, taxonomy])
   149→
   150→  const categories = useMemo(() => {
   151→    const inStock = [
   152→      ...new Set((data?.vehicles ?? []).map((v) => v.category).filter((c): c is string => !!c)),
   153→    ].sort()
   154→    return [...new Set([...(taxonomy?.categories ?? []), ...inStock])]
   155→  }, [data, taxonomy])
   156→
   157→  // Cabang aktif — UI lokasi (badge, filter, Maps) otomatis hilang bila kosong
   158→  const branches = data?.branches ?? []
   159→  const hasBranches = branches.length > 0
   160→
   161→  const filtered = useMemo(() => {
   162→    if (!data) return []
   163→    const query = q.trim().toLowerCase()
   164→    return data.vehicles.filter((v) => {
   165→      if (statusFilter === 'available' && v.status !== 'available') return false
   166→      if (statusFilter === 'hold' && v.status !== 'hold') return false
   167→      if (brandFilter !== 'all' && v.brand !== brandFilter) return false
   168→      if (categoryFilter !== 'all' && v.category !== categoryFilter) return false
   169→      if (branchFilter === 'main' && v.branch) return false
   170→      if (branchFilter !== 'all' && branchFilter !== 'main' && v.branch?.id !== branchFilter)
   171→        return false
   172→      if (!query) return true
   173→      return [v.brand, v.model, v.licensePlate, String(v.year)]
   174→        .join(' ')
   175→        .toLowerCase()
   176→        .includes(query)
   177→    })
   178→  }, [data, statusFilter, brandFilter, categoryFilter, branchFilter, q])
   179→
   180→  async function handleCopy(v: PublicVehicle) {
   181→    if (!data) return
   182→    const ok = await copyToClipboard(buildAdText(v, data.showroom))
   183→    if (ok) toast.success('Iklan disalin! Tinggal paste ke WhatsApp Status / Marketplace.')
   184→    else toast.error('Gagal menyalin. Coba lagi.')
   185→  }
   186→
   187→  /** Verifikasi rekanan sukses — simpan sesi di localStorage lalu buka katalog. */
   188→  function handleVerified(s: MarketingSession) {
   189→    saveMarketingSession(slug, s)
   190→    marketingRef.current = s
   191→    setMarketing(s)
   192→    setPhase('open')
   193→    refetch(false, s)
   194→  }
   195→
   196→  /** Keluar / ganti nomor WhatsApp. */
   197→  function handleChangeNumber() {
   198→    clearMarketingSession(slug)
   199→    marketingRef.current = null
   200→    setMarketing(null)
   201→    setData(null)
   202→    setPhase('gate')
   203→  }
   204→
   205→  if (notFound) {
   206→    return (
   207→      <div className="mx-auto flex w-full max-w-md flex-1 flex-col items-center justify-center px-4 py-16 text-center">
   208→        <Camera className="h-10 w-10 text-slate-300" aria-hidden />
   209→        <h1 className="mt-3 text-lg font-extrabold text-slate-900">Katalog tidak ditemukan</h1>
   210→        <p className="mt-1 text-sm text-slate-500">
   211→          Katalog dengan alamat <code className="font-bold">/s/{slug}</code> tidak terdaftar
   212→          atau sudah dinonaktifkan.
   213→        </p>
   214→      </div>
   215→    )
   216→  }
   217→
   218→  // ============ Gerbang rekanan: layar verifikasi nomor WhatsApp ============
   219→  if (phase === 'gate') {
   220→    return <MarketingGate slug={slug} onVerified={handleVerified} />
   221→  }
   222→
   223→  // Fase checking: skeleton layar penuh (hindari kedipan katalog sebelum gate diputuskan)
   224→  if (phase === 'checking') {
   225→    return (
   226→      <div className="mx-auto w-full max-w-md flex-1 space-y-3 px-4 py-10">
   227→        <div className="h-40 animate-pulse rounded-lg bg-slate-200" />
   228→      </div>
   229→    )
   230→  }
   231→
   232→  const showroom = data?.showroom
   233→
   234→  return (
   235→    <>
   236→      {/* Header katalog */}
   237→      <header className="border-b border-slate-200 bg-white">
   238→        <div className="mx-auto flex h-14 w-full max-w-5xl items-center gap-3 px-4">
   239→          {showroom?.logoUrl && (
   240→            <img
   241→              src={showroom.logoUrl}
   242→              alt={`Logo ${showroom.name}`}
   243→              className="h-10 w-10 shrink-0 rounded-md object-cover ring-1 ring-slate-200"
   244→            />
   245→          )}
   246→          <div className="min-w-0 flex-1">
   247→            <h1 className="truncate text-sm font-extrabold text-slate-900">
   248→              {showroom?.name ?? 'Memuat katalog...'}
   249→            </h1>
   250→            <p className="truncate text-xs text-slate-500">{showroom?.address ?? '\u00A0'}</p>
   251→          </div>
   252→          {showroom?.mapsUrl && (
   253→            <a
   254→              href={showroom.mapsUrl}
   255→              target="_blank"
   256→              rel="noreferrer"
   257→              className="inline-flex h-10 shrink-0 items-center gap-1.5 rounded-lg border border-slate-300 bg-white px-3 text-xs font-extrabold text-slate-700 hover:bg-slate-50"
   258→              title="Buka lokasi di Google Maps"
   259→            >
   260→              <MapPin className="h-4 w-4" aria-hidden />
   261→              <span className="hidden sm:inline">Maps</span>
   262→            </a>
   263→          )}
   264→          {showroom && (
   265→            <a
   266→              href={waLink(
   267→                showroom.ownerPhone,
   268→                `Halo ${showroom.name}, saya lihat katalog OtoStok Anda. Ada unit yang menarik.`,
   269→              )}
   270→              target="_blank"
   271→              rel="noreferrer"
   272→              className="inline-flex h-10 shrink-0 items-center gap-1.5 rounded-lg bg-emerald-700 px-3 text-xs font-extrabold text-white hover:bg-emerald-800"
   273→            >
   274→              <MessageCircle className="h-4 w-4" aria-hidden />
   275→              <span className="hidden sm:inline">Chat Showroom</span>
   276→            </a>
   277→          )}
   278→        </div>
   279→        {/* Foto header katalog (diatur owner di Pengaturan Showroom) */}
   280→        {showroom?.headerUrl && (
   281→          <div className="mx-auto max-h-44 w-full max-w-5xl overflow-hidden sm:max-h-56">
   282→            <img
   283→              src={showroom.headerUrl}
   284→              alt={`Foto header ${showroom.name}`}
   285→              className="h-44 w-full object-cover sm:h-56"
   286→            />
   287→          </div>
   288→        )}
   289→        {/* Sapaan personal rekanan — hanya bila showroom memakai whitelist */}
   290→        {marketing && (
   291→          <div className="mx-auto w-full max-w-5xl border-t border-slate-200 bg-blue-50">
   292→            <div className="flex h-10 items-center justify-between gap-2 px-4">
   293→              <p className="truncate text-xs font-extrabold text-blue-900">
   294→                <BadgeCheck className="mr-1 inline h-3.5 w-3.5 text-blue-700" aria-hidden />
   295→                Halo, {marketing.fullName} ({marketing.addressCity})
   296→              </p>
   297→              <button
   298→                type="button"
   299→                onClick={handleChangeNumber}
   300→                className="shrink-0 text-[11px] font-bold text-blue-700 underline hover:text-blue-900"
   301→              >
   302→                Ganti Nomor
   303→              </button>
   304→            </div>
   305→          </div>
   306→        )}
   307→      </header>
   308→
   309→      {/* Filter */}
   310→      <div className="border-b border-slate-200 bg-white">
   311→        <div className="mx-auto w-full max-w-5xl space-y-2 px-4 py-3">
   312→          <div className="flex items-center gap-2">
   313→            <div className="relative flex-1">
   314→              <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-slate-400" />
   315→              <Input
   316→                value={q}
   317→                onChange={(e) => setQ(e.target.value)}
   318→                placeholder="Cari model atau plat nomor..."
   319→                className="h-10 pl-9"
   320→                aria-label="Cari unit di katalog"
   321→              />
   322→            </div>
   323→            {data && (
   324→              <span className="hidden shrink-0 text-[11px] text-slate-400 sm:block">
   325→                Diperbarui {formatDateTimeID(data.fetchedAt)}
   326→              </span>
   327→            )}
   328→          </div>
   329→
   330→          <div className="flex flex-wrap items-center gap-2">
   331→            <div className="flex rounded-lg border border-slate-300 bg-white p-0.5">
   332→              <button
   333→                type="button"
   334→                onClick={() => setStatusFilter('all')}
   335→                aria-pressed={statusFilter === 'all'}
   336→                className={`h-9 rounded-md px-3 text-xs font-bold ${
   337→                  statusFilter === 'all'
   338→                    ? 'bg-slate-900 text-white'
   339→                    : 'text-slate-600 hover:bg-slate-100'
   340→                }`}
   341→              >
   342→                Semua{data ? ` (${data.vehicles.length})` : ''}
   343→              </button>
   344→              <button
   345→                type="button"
   346→                onClick={() => setStatusFilter('available')}
   347→                aria-pressed={statusFilter === 'available'}
   348→                className={`h-9 rounded-md px-3 text-xs font-bold ${
   349→                  statusFilter === 'available'
   350→                    ? 'bg-emerald-600 text-white'
   351→                    : 'text-slate-600 hover:bg-slate-100'
   352→                }`}
   353→              >
   354→                Ready{data ? ` (${data.counts.available})` : ''}
   355→              </button>
   356→              <button
   357→                type="button"
   358→                onClick={() => setStatusFilter('hold')}
   359→                aria-pressed={statusFilter === 'hold'}
   360→                className={`h-9 rounded-md px-3 text-xs font-bold ${
   361→                  statusFilter === 'hold'
   362→                    ? 'bg-amber-500 text-white'
   363→                    : 'text-slate-600 hover:bg-slate-100'
   364→                }`}
   365→              >
   366→                Ditahan{data ? ` (${data.counts.hold})` : ''}
   367→              </button>
   368→            </div>
   369→          </div>
   370→
   371→          {/* Chip kategori */}
   372→          {categories.length > 0 && (
   373→            <div className="scrollbar-thin flex gap-1.5 overflow-x-auto pb-1" role="group" aria-label="Filter kategori">
   374→              <button
   375→                type="button"
   376→                onClick={() => setCategoryFilter('all')}
   377→                className={`h-8 shrink-0 rounded-md border px-3 text-xs font-bold ${
   378→                  categoryFilter === 'all'
   379→                    ? 'border-blue-700 bg-blue-700 text-white'
   380→                    : 'border-slate-300 bg-white text-slate-600'
   381→                }`}
   382→              >
   383→                Semua Kategori
   384→              </button>
   385→              {categories.map((c) => (
   386→                <button
   387→                  key={c}
   388→                  type="button"
   389→                  onClick={() => setCategoryFilter(c)}
   390→                  className={`h-8 shrink-0 rounded-md border px-3 text-xs font-bold ${
   391→                    categoryFilter === c
   392→                      ? 'border-blue-700 bg-blue-700 text-white'
   393→                      : 'border-slate-300 bg-white text-slate-600'
   394→                  }`}
   395→                >
   396→                  {c}
   397→                </button>
   398→              ))}
   399→            </div>
   400→          )}
   401→
   402→          {brands.length > 1 && (
   403→            <div className="scrollbar-thin flex gap-1.5 overflow-x-auto pb-1">
   404→              <button
   405→                type="button"
   406→                onClick={() => setBrandFilter('all')}
   407→                className={`h-8 shrink-0 rounded-md border px-3 text-xs font-bold ${
   408→                  brandFilter === 'all'
   409→                    ? 'border-slate-900 bg-slate-900 text-white'
   410→                    : 'border-slate-300 bg-white text-slate-600'
   411→                }`}
   412→              >
   413→                Semua Merek
   414→              </button>
   415→              {brands.map((b) => (
   416→                <button
   417→                  key={b}
   418→                  type="button"
   419→                  onClick={() => setBrandFilter(b)}
   420→                  className={`h-8 shrink-0 rounded-md border px-3 text-xs font-bold ${
   421→                    brandFilter === b
   422→                      ? 'border-slate-900 bg-slate-900 text-white'
   423→                      : 'border-slate-300 bg-white text-slate-600'
   424→                  }`}
   425→                >
   426→                  {b}
   427→                </button>
   428→              ))}
   429→            </div>
   430→          )}
   431→
   432→          {/* Chip lokasi cabang — hanya tampil bila showroom punya cabang */}
   433→          {hasBranches && (
   434→            <div
   435→              className="scrollbar-thin flex gap-1.5 overflow-x-auto pb-1"
   436→              role="group"
   437→              aria-label="Filter lokasi cabang"
   438→            >
   439→              <button
   440→                type="button"
   441→                onClick={() => setBranchFilter('all')}
   442→                className={`h-8 shrink-0 rounded-md border px-3 text-xs font-bold ${
   443→                  branchFilter === 'all'
   444→                    ? 'border-emerald-700 bg-emerald-700 text-white'
   445→                    : 'border-slate-300 bg-white text-slate-600'
   446→                }`}
   447→              >
   448→                Semua Lokasi
   449→              </button>
   450→              <button
   451→                type="button"
   452→                onClick={() => setBranchFilter('main')}
   453→                className={`h-8 shrink-0 rounded-md border px-3 text-xs font-bold ${
   454→                  branchFilter === 'main'
   455→                    ? 'border-emerald-700 bg-emerald-700 text-white'
   456→                    : 'border-slate-300 bg-white text-slate-600'
   457→                }`}
   458→              >
   459→                Lokasi Utama
   460→              </button>
   461→              {branches.map((b) => (
   462→                <button
   463→                  key={b.id}
   464→                  type="button"
   465→                  onClick={() => setBranchFilter(b.id)}
   466→                  className={`h-8 shrink-0 rounded-md border px-3 text-xs font-bold ${
   467→                    branchFilter === b.id
   468→                      ? 'border-emerald-700 bg-emerald-700 text-white'
   469→                      : 'border-slate-300 bg-white text-slate-600'
   470→                  }`}
   471→                >
   472→                  {b.name}
   473→                </button>
   474→              ))}
   475→            </div>
   476→          )}
   477→        </div>
   478→      </div>
   479→
   480→      {/* Daftar unit */}
   481→      <div className="mx-auto w-full max-w-5xl flex-1 px-4 py-4">
   482→        {loading && (
   483→          <div className="grid gap-3 sm:grid-cols-2 lg:grid-cols-3">
   484→            {[0, 1, 2, 3, 4, 5].map((i) => (
   485→              <div
   486→                key={i}
   487→                className="overflow-hidden rounded-lg border border-slate-200 bg-white"
   488→              >
   489→                <div className="aspect-[4/3] animate-pulse bg-slate-200" />
   490→                <div className="space-y-2 p-3">
   491→                  <div className="h-4 w-3/4 animate-pulse rounded bg-slate-200" />
   492→                  <div className="h-3 w-1/2 animate-pulse rounded bg-slate-200" />
   493→                  <div className="h-6 w-2/3 animate-pulse rounded bg-slate-200" />
   494→                </div>
   495→              </div>
   496→            ))}
   497→          </div>
   498→        )}
   499→
   500→        {error && !loading && (
   501→          <div className="rounded-lg border border-red-200 bg-red-50 p-4 text-center">
   502→            <p className="text-sm font-bold text-red-800">{error}</p>
   503→            <Button
   504→              variant="outline"
   505→              className="mt-3 h-10 border-red-300 text-xs font-bold"
   506→              onClick={() => refetch()}
   507→            >
   508→              Coba Lagi
   509→            </Button>
   510→          </div>
   511→        )}
   512→
   513→        {data && !loading && filtered.length === 0 && (
   514→          <div className="flex flex-col items-center rounded-lg border border-dashed border-slate-300 bg-white px-4 py-12 text-center">
   515→            <Search className="h-8 w-8 text-slate-300" aria-hidden />
   516→            <p className="mt-2 text-sm font-bold text-slate-700">
   517→              {data.vehicles.length === 0
   518→                ? 'Belum ada unit di katalog ini.'
   519→                : 'Tidak ada unit yang cocok dengan filter.'}
   520→            </p>
   521→          </div>
   522→        )}
   523→
   524→        {data && !loading && filtered.length > 0 && (
   525→          <div className="grid gap-3 sm:grid-cols-2 lg:grid-cols-3">
   526→            {filtered.map((v) => (
   527→              <CatalogCard
   528→                key={v.id}
   529→                v={v}
   530→                showLocation={hasBranches}
   531→                fallbackLocName={showroom?.name ?? ''}

   577→                                  (g.index - 1 + g.vehicle.photos.length) %

   614→            </div>
   615→          )}

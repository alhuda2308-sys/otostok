    44→/** Sesi Super Admin disimpan di sessionStorage (hilang saat tab ditutup). */
    45→const SESSION_KEY = 'otostok_sa_key'
    46→const DAY_MS = 24 * 60 * 60 * 1000
    47→/** Ambang "segera kedaluwarsa" untuk badge kuning. */
    48→const NEAR_EXPIRY_MS = 7 * DAY_MS
    49→
    50→interface SuperLicenseRow {
    51→  id: string
    52→  licenseKey: string
    53→  planType: string
    54→  maxVehicles: number
    55→  status: string
    56→  expiresAt: string | null
    57→  createdAt: string
    58→  showroom: {
    59→    id: string
    60→    name: string
    61→    slug: string
    62→    ownerPhone: string
    63→    isActive: boolean
    64→  } | null
    65→  usedUnits: number
    66→  totalUnits: number
    67→}
    68→
    69→type EffectiveStatus = 'active' | 'near_expiry' | 'expired' | 'suspended'
    70→
    71→/** Status efektif: raw DB + kedaluwarsa lazily dari expiresAt. */
    72→function effectiveStatus(row: SuperLicenseRow): EffectiveStatus {
    73→  if (row.status === 'suspended') return 'suspended'
    74→  if (row.expiresAt && new Date(row.expiresAt).getTime() < Date.now()) return 'expired'
    75→  if (row.expiresAt && new Date(row.expiresAt).getTime() - Date.now() <= NEAR_EXPIRY_MS)
    76→    return 'near_expiry'
    77→  return 'active'
    78→}
    79→
    80→const STATUS_BADGE: Record<EffectiveStatus, { label: string; className: string }> = {
    81→  active: { label: 'Aktif', className: 'bg-emerald-600 text-white' },
    82→  near_expiry: { label: 'Segera Habis', className: 'bg-amber-500 text-white' },
    83→  expired: { label: 'Kedaluwarsa', className: 'bg-red-600 text-white' },
    84→  suspended: { label: 'Suspended', className: 'bg-red-800 text-white' },
    85→}
    86→
    87→function expiryLabel(row: SuperLicenseRow): string {
    88→  if (!row.expiresAt) return 'Lifetime'
    89→  return `s/d ${formatDateID(row.expiresAt)}`
    90→}
    91→
    92→function planLabel(planType: string): string {
    93→  const plan = SUPER_PLANS.find((p) => p.value === planType)
    94→  return plan?.label ?? PLAN_LABELS[planType] ?? planType
    95→}

    97→/** Template pesan WA siap kirim ke pembeli lisensi. */
    98→function buildBuyerWaMessage(license: SuperLicenseRow): string {
    99→  const lines: string[] = []
   100→  lines.push('*OtoStok — AKTIVASI LISENSI*')
   101→  lines.push('')
   102→  lines.push(`Kode Lisensi : *${license.licenseKey}*`)
   103→  lines.push(`Paket : ${planLabel(license.planType)}`)
   104→  lines.push(`Kuota Unit : ${license.maxVehicles} motor`)
   105→  lines.push(
   106→    license.expiresAt
   107→      ? `Masa Aktif : ${expiryLabel(license)}`
   108→      : 'Masa Aktif : Lifetime (tanpa batas waktu)',
   109→  )
   110→  lines.push('')
   111→  lines.push(`Aktivasi : ${window.location.origin}/activate`)
   112→  lines.push('')
   113→  lines.push('Cara aktivasi:')
   114→  lines.push('1. Buka link di atas')
   115→  lines.push('2. Masukkan kode lisensi')
   116→  lines.push('3. Lengkapi data showroom & buat password owner')
   117→  lines.push('4. Dashboard + katalog marketing langsung siap dipakai')
   118→  lines.push('')
   119→  lines.push('Butuh bantuan? Balas pesan ini.')
   120→  return lines.join('\n')
   121→}

   123→/** Ringkasan data lisensi utk tombol "Salin Info". */
   124→function buildRowSummary(row: SuperLicenseRow): string {
   125→  const st = STATUS_BADGE[effectiveStatus(row)].label
   126→  const lines: string[] = []
   127→  lines.push(`Lisensi: ${row.licenseKey} (${planLabel(row.planType)})`)
   128→  lines.push(`Status: ${st}`)
   129→  lines.push(
   130→    row.showroom
   131→      ? `Showroom: ${row.showroom.name} — /s/${row.showroom.slug}`
   132→      : 'Showroom: belum diaktivasi',
   133→  )
   134→  if (row.showroom) {
   135→    lines.push(`Owner WA: ${formatPhoneDisplay(row.showroom.ownerPhone)}`)
   136→    lines.push(`Kuota: ${row.usedUnits}/${row.maxVehicles} unit (total tercatat ${row.totalUnits})`)
   137→  }
   138→  lines.push(`Masa aktif: ${expiryLabel(row)}`)
   139→  lines.push(`Dibuat: ${formatDateID(row.createdAt)}`)
   140→  return lines.join('\n')
   141→}

   161→  const secretRef = useRef<string | null>(null)
   162→
   163→  const refetch = useCallback(
   164→    async (secret?: string) => {
   165→      const key = secret ?? secretRef.current
   166→      if (!key) return
   167→      setRefreshing(true)
   168→      try {
   169→        const res = await fetch('/api/super-admin/licenses', {
   170→          headers: { 'X-Super-Secret': key },
   171→          cache: 'no-store',
   172→        })
   173→        if (res.status === 401) {
   174→          // Kunci dicabut / salah — paksa login ulang
   175→          sessionStorage.removeItem(SESSION_KEY)
   176→          secretRef.current = null
   177→          setPhase('login')
   178→          setAuthError('Sesi tidak valid lagi. Masukkan Master Secret Key.')
   179→          return
   180→        }
   181→        if (!res.ok) {
   182→          const j = await res.json().catch(() => ({}))
   183→          throw new Error(j.error || 'Gagal memuat data lisensi.')
   184→        }
   185→        const j = await res.json()
   186→        setRows(j.licenses)
   187→      } catch (e) {
   188→        toast.error(e instanceof Error ? e.message : 'Gagal memuat data lisensi.')
   189→      } finally {
   190→        setRefreshing(false)
   191→      }
   192→    },
   193→    [],
   194→  )

   196→  /** Verifikasi passkey ke server; sekaligus memuat data pertama. */
   197→  async function verifyAndLoad(secret: string) {
   198→    setChecking(true)
   199→    setAuthError(null)
   200→    try {
   201→      const res = await fetch('/api/super-admin/licenses', {
   202→        headers: { 'X-Super-Secret': secret },
   203→        cache: 'no-store',
   204→      })
   205→      if (res.status === 401) {
   206→        setAuthError('Master Secret Key salah.')
   207→        sessionStorage.removeItem(SESSION_KEY)
   208→        return
   209→      }
   210→      if (!res.ok) {
   211→        const j = await res.json().catch(() => ({}))
   212→        throw new Error(j.error || 'Gagal verifikasi kunci.')
   213→      }
   214→      const j = await res.json()
   215→      secretRef.current = secret
   216→      sessionStorage.setItem(SESSION_KEY, secret)
   217→      setRows(j.licenses)
   218→      setPhase('ready')
   219→      setPasskey('')
   220→    } catch (e) {
   221→      setAuthError(e instanceof Error ? e.message : 'Gagal verifikasi kunci.')
   222→    } finally {
   223→      setChecking(false)
   224→    }
   225→  }
   226→
   227→  // Auto-login dari sessionStorage — cukup sekali per tab (bukan dependency render)
   228→  useEffect(() => {
   229→    const saved = sessionStorage.getItem(SESSION_KEY)
   230→    if (saved) void verifyAndLoad(saved)
   231→  }, [])

   233→  function handleLogin(e: React.FormEvent) {
   234→    e.preventDefault()
   235→    const key = passkey.trim()
   236→    if (!key) {
   237→      setAuthError('Masukkan Master Secret Key.')
   238→      return
   239→    }
   240→    void verifyAndLoad(key)
   241→  }
   242→
   243→  function handleLogout() {
   244→    sessionStorage.removeItem(SESSION_KEY)
   245→    secretRef.current = null
   246→    setRows([])
   247→    setResult(null)
   248→    setPasskey('')
   249→    setPhase('login')
   250→  }

   252→  /** Generate lisensi baru 1-klik. */
   253→  async function handleGenerate(e: React.FormEvent) {
   254→    e.preventDefault()
   255→    const quota = Number(maxVehicles)
   256→    if (!Number.isInteger(quota) || quota < 1 || quota > 10000) {
   257→      toast.error('Kuota unit harus angka 1 - 10.000.')
   258→      return
   259→    }
   260→    setGenerating(true)
   261→    try {
   262→      const res = await fetch('/api/super-admin/licenses', {
   263→        method: 'POST',
   264→        headers: {
   265→          'Content-Type': 'application/json',
   266→          'X-Super-Secret': secretRef.current ?? '',
   267→        },
   268→        body: JSON.stringify({ planType, maxVehicles: quota }),
   269→      })
   270→      if (res.status === 401) {
   271→        await refetch()
   272→        return
   273→      }
   274→      const j = await res.json()
   275→      if (!res.ok) throw new Error(j.error || 'Gagal membuat lisensi.')
   276→      const license = { ...(j.license as SuperLicenseRow), showroom: null, usedUnits: 0, totalUnits: 0 }
   277→      setResult(license)
   278→      toast.success(`Lisensi ${license.licenseKey} berhasil dibuat!`)
   279→      await refetch()
   280→    } catch (e) {
   281→      toast.error(e instanceof Error ? e.message : 'Gagal membuat lisensi.')
   282→    } finally {
   283→      setGenerating(false)
   284→    }
   285→  }

   287→  /** Aksi cepat per baris: extend / suspend / unsuspend. */
   288→  async function handleAction(row: SuperLicenseRow, action: 'extend' | 'suspend' | 'unsuspend') {
   289→    setBusyId(`${row.id}-${action}`)
   290→    try {
   291→      const res = await fetch('/api/super-admin/licenses', {
   292→        method: 'PATCH',
   293→        headers: {
   294→          'Content-Type': 'application/json',
   295→          'X-Super-Secret': secretRef.current ?? '',
   296→        },
   297→        body: JSON.stringify({ id: row.id, action }),
   298→      })
   299→      if (res.status === 401) {
   300→        await refetch()
   301→        return
   302→      }
   303→      const j = await res.json()
   304→      if (!res.ok) throw new Error(j.error || 'Aksi gagal.')
   305→      if (action === 'extend') toast.success(`+30 hari — masa aktif kini ${expiryLabel(row)}`)
   306→      if (action === 'suspend') toast.success(`${row.licenseKey} dibekukan (suspend).`)
   307→      if (action === 'unsuspend') toast.success(`${row.licenseKey} dibuka kembali.`)
   308→      await refetch()
   309→    } catch (e) {
   310→      toast.error(e instanceof Error ? e.message : 'Aksi gagal.')
   311→    } finally {
   312→      setBusyId(null)
   313→    }
   314→  }

   524→              <div className="mt-3 grid grid-cols-2 gap-2">
   525→                <Button

   541→            </div>
   542→          )}
   543→        </section>
   544→
   545→        {/* Tabel monitoring */}
   546→        <section className="rounded-lg border border-slate-200 bg-white lg:col-span-3">
   547→          <div className="flex flex-wrap items-center gap-2 border-b border-slate-200 p-4">

   550→            </h2>
   551→            <Button
   552→              variant="outline"
   553→              size="sm"

   555→              onClick={() => refetch()}
   556→              disabled={refreshing}
   557→            >
   558→              <RefreshCcw
   559→                className={`mr-1 h-3.5 w-3.5 ${refreshing ? 'animate-spin' : ''}`}
   560→                aria-hidden
   561→              />
   562→              Muat Ulang
   563→            </Button>
   564→          </div>
   565→          <div className="border-b border-slate-200 px-4 py-3">
   566→            <div className="relative">
   567→              <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-slate-400" />

   574→              />

   580→              {rows.length === 0 ? 'Belum ada lisensi terdaftar.' : 'Tidak ada hasil yang cocok.'}
   581→            </p>
   582→          ) : (
   583→            <>
   584→              {/* Tabel desktop */}
   585→              <div className="hidden max-h-[36rem] overflow-y-auto lg:block">
   586→                <Table>
   587→                  <TableHeader className="sticky top-0 z-10 bg-white shadow-[0_1px_0_0_#e2e8f0]">

   604→                        onAction={handleAction}

   612→
   613→              {/* Kartu mobile */}
   614→              <div className="max-h-[40rem] space-y-3 overflow-y-auto p-4 lg:hidden">

   620→                    onAction={handleAction}
   621→                    onCopy={copy}
   622→                    compact
   623→                  />
   624→                ))}
   625→              </div>
   626→            </>
   627→          )}
   628→        </section>
   629→      </div>
   630→    </div>
   631→  )
   632→}

   635→function LicenseRowActions({
   636→  row,
   637→  busyId,
   638→  onAction,
   639→  onCopy,
   640→  compact,
   641→}: {

   645→  onCopy: (text: string, okMsg: string) => void
   646→  compact: boolean
   647→}) {
   648→  const st = effectiveStatus(row)

   663→    </div>

   665→
   666→  const showroomCell = row.showroom ? (
   667→    <div className="min-w-0">

   712→        {row.usedUnits} / {row.maxVehicles} <span className="font-semibold">unit</span>

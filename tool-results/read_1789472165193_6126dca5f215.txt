    59→    59→  // Sistem Rekanan Terdaftar: sesi verifikasi WhatsApp marketing

    66→    66→  const [categoryFilter, setCategoryFilter] = useState<string>('all')

    68→    68→  const [q, setQ] = useState('')

    72→    72→
    73→    73→  const refetch = useCallback(

    76→    76→      try {
    77→    77→        const mkt = session ?? marketingRef.current
    78→    78→        const res = await fetch(`/api/showrooms/${slug}/vehicles`, {
    79→    79→          cache: 'no-store',
    80→    80→          ...(mkt ? { headers: { 'X-Mkt-Phone': mkt.phone } } : {}),
    81→    81→        })
    82→    82→        if (res.status === 404) {
    83→    83→          setNotFound(true)
    84→    84→          return
    85→    85→        }

    87→    87→          const j = await res.json().catch(() => ({}))
    88→    88→          if (j.code === 'WHITELIST_REQUIRED') {
    89→    89→            // Nomor sesi tidak lagi terdaftar/aktif — paksa verifikasi ulang
    90→    90→            clearMarketingSession(slug)
    91→    91→            marketingRef.current = null
    92→    92→            setMarketing(null)
    93→    93→            setPhase('gate')
    94→    94→            return
    95→    95→          }
    96→    96→          throw new Error(j.error || 'Akses katalog ditolak.')
    97→    97→        }
    98→    98→        if (!res.ok) {
    99→    99→          const j = await res.json().catch(() => ({}))
   100→   100→          throw new Error(j.error || 'Gagal memuat katalog.')
   101→   101→        }
   102→   102→        setData(await res.json())
   103→   103→        setError(null)
   104→   104→        setPhase('open')
   105→   105→      } catch (e) {
   106→   106→        if (!silent) setError(e instanceof Error ? e.message : 'Gagal memuat katalog.')
   107→   107→      } finally {
   108→   108→        if (!silent) setLoading(false)
   109→   109→      }
   110→   110→    },
   111→   111→    [slug],
   112→   112→  )
   113→   113→
   114→   114→  useEffect(() => {
   115→   115→    // Baca sesi rekanan dari localStorage — marketing tidak perlu mengetik ulang nomor

   118→   118→    setMarketing(cached)
   119→   119→    refetch(false, cached)
   120→   120→  }, [refetch, slug])
   121→   121→

   137→   137→    return () => {
   138→   138→      clearInterval(iv)
   139→   139→      document.removeEventListener('visibilitychange', onVis)
   140→   140→    }
   141→   141→  }, [refetch])
   142→   142→

   152→   152→      ...new Set((data?.vehicles ?? []).map((v) => v.category).filter((c): c is string => !!c)),
   153→   153→    ].sort()
   154→   154→    return [...new Set([...(taxonomy?.categories ?? []), ...inStock])]
   155→   155→  }, [data, taxonomy])
   156→   156→
   157→   157→  // Cabang aktif — UI lokasi (badge, filter, Maps) otomatis hilang bila kosong
   158→   158→  const branches = data?.branches ?? []
   159→   159→  const hasBranches = branches.length > 0
   160→   160→
   161→   161→  const filtered = useMemo(() => {
   162→   162→    if (!data) return []
   163→   163→    const query = q.trim().toLowerCase()
   164→   164→    return data.vehicles.filter((v) => {
   165→   165→      if (statusFilter === 'available' && v.status !== 'available') return false
   166→   166→      if (statusFilter === 'hold' && v.status !== 'hold') return false
   167→   167→      if (brandFilter !== 'all' && v.brand !== brandFilter) return false
   168→   168→      if (categoryFilter !== 'all' && v.category !== categoryFilter) return false
   169→   169→      if (branchFilter === 'main' && v.branch) return false
   170→   170→      if (branchFilter !== 'all' && branchFilter !== 'main' && v.branch?.id !== branchFilter)
   171→   171→        return false
   172→   172→      if (!query) return true
   173→   173→      return [v.brand, v.model, v.licensePlate, String(v.year)]
   174→   174→        .join(' ')
   175→   175→        .toLowerCase()
   176→   176→        .includes(query)
   177→   177→    })
   178→   178→  }, [data, statusFilter, brandFilter, categoryFilter, branchFilter, q])
   179→   179→

   210→   210→        <p className="mt-1 text-sm text-slate-500">
   211→   211→          Katalog dengan alamat <code className="font-bold">/s/{slug}</code> tidak terdaftar
   212→   212→          atau sudah dinonaktifkan.
   213→   213→        </p>
   214→   214→      </div>
   215→   215→    )
   216→   216→  }
   217→   217→
   218→   218→  // ============ Gerbang rekanan: layar verifikasi nomor WhatsApp ============

   220→   220→    return <MarketingGate slug={slug} onVerified={handleVerified} />
   221→   221→  }
   222→   222→
   223→   223→  // Fase checking: skeleton layar penuh (hindari kedipan katalog sebelum gate diputuskan)

   225→   225→    return (
   226→   226→      <div className="mx-auto w-full max-w-md flex-1 space-y-3 px-4 py-10">
   227→   227→        <div className="h-40 animate-pulse rounded-lg bg-slate-200" />
   228→   228→      </div>

   230→   230→  }
   231→   231→
   232→   232→  const showroom = data?.showroom
   233→   233→
   234→   234→  return (
   235→   235→    <>
   236→   236→      {/* Header katalog */}
   237→   237→      <header className="border-b border-slate-200 bg-white">
   238→   238→        <div className="mx-auto flex h-14 w-full max-w-5xl items-center gap-3 px-4">
   239→   239→          {showroom?.logoUrl && (
   240→   240→            <img
   241→   241→              src={showroom.logoUrl}
   242→   242→              alt={`Logo ${showroom.name}`}
   243→   243→              className="h-10 w-10 shrink-0 rounded-md object-cover ring-1 ring-slate-200"

   246→   246→          <div className="min-w-0 flex-1">
   247→   247→            <h1 className="truncate text-sm font-extrabold text-slate-900">
   248→   248→              {showroom?.name ?? 'Memuat katalog...'}
   249→   249→            </h1>
   250→   250→            <p className="truncate text-xs text-slate-500">{showroom?.address ?? '\u00A0'}</p>
   251→   251→          </div>
   252→   252→          {showroom?.mapsUrl && (
   253→   253→            <a
   254→   254→              href={showroom.mapsUrl}
   255→   255→              target="_blank"
   256→   256→              rel="noreferrer"
   257→   257→              className="inline-flex h-10 shrink-0 items-center gap-1.5 rounded-lg border border-slate-300 bg-white px-3 text-xs font-extrabold text-slate-700 hover:bg-slate-50"
   258→   258→              title="Buka lokasi di Google Maps"
   259→   259→            >

   270→   270→              target="_blank"

   299→   299→                onClick={handleChangeNumber}

   305→   305→          </div>

   309→   309→      {/* Filter */}

   311→   311→        <div className="mx-auto w-full max-w-5xl space-y-2 px-4 py-3">
   312→   312→          <div className="flex items-center gap-2">

   315→   315→              <Input

   321→   321→              />
   322→   322→            </div>
   323→   323→            {data && (
   324→   324→              <span className="hidden shrink-0 text-[11px] text-slate-400 sm:block">
   325→   325→                Diperbarui {formatDateTimeID(data.fetchedAt)}
   326→   326→              </span>
   327→   327→            )}
   328→   328→          </div>
   329→   329→
   330→   330→          <div className="flex flex-wrap items-center gap-2">
   331→   331→            <div className="flex rounded-lg border border-slate-300 bg-white p-0.5">
   332→   332→              <button

   334→   334→                onClick={() => setStatusFilter('all')}

   431→   431→
   432→   432→          {/* Chip lokasi cabang — hanya tampil bila showroom punya cabang */}
   433→   433→          {hasBranches && (
